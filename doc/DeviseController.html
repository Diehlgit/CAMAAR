def find_message(kind, options = {})
  options[:scope] ||= translation_scope
  options[:default] = Array(options[:default]).unshift(kind.to_sym)
  options[:resource_name] = resource_name
  options = devise_i18n_options(options)
  I18n.t("#{options[:resource_name]}.#{kind}", **options)
end

    # File camaar/app/controllers/devise_controller.rb, line 106
def require_no_authentication
  assert_is_devise_resource!
  return unless is_navigational_format?
  no_input = devise_mapping.no_input_strategies

  authenticated = if no_input.present?
    args = no_input.dup.push scope: resource_name
    warden.authenticate?(*args)
  else
    warden.authenticated?(resource_name)
  end

  if authenticated && resource = warden.user(resource_name)
    set_flash_message(:alert, 'already_authenticated', scope: 'devise.failure')
    redirect_to after_sign_in_path_for(resource)
  end
end

def set_flash_message(key, kind, options = {})
  message = find_message(kind, options)
  if options[:now]
    flash.now[key] = message if message.present?
  else
    flash[key] = message if message.present?
  end
end

    successfully_sent?(resource) click to toggle source

    Método helper para uso após chamar métodos send_*_instructions em um recurso.

    # File camaar/app/controllers/devise_controller.rb, line 125
def successfully_sent?(resource)
  notice = if Devise.paranoid
    resource.errors.clear
    :send_paranoid_instructions
  elsif resource.errors.empty?
    :send_instructions
  end

  if notice
    set_flash_message! :notice, notice
    true
  end
end
